cmake_minimum_required(VERSION 3.16)
project(rvclaw_depth_assist LANGUAGES CXX)
find_path(OPENNI2_INCLUDE_DIR OpenNI.h PATH_SUFFIXES openni2 REQUIRED)
find_library(OPENNI2_LIBRARY NAMES OpenNI2 REQUIRED)
add_executable(rvclaw_depth_reader openni_depth_reader.cpp)
target_compile_features(rvclaw_depth_reader PRIVATE cxx_std_17)
target_compile_options(rvclaw_depth_reader PRIVATE -Wall -Wextra -Werror)
target_include_directories(rvclaw_depth_reader PRIVATE ${OPENNI2_INCLUDE_DIR})
target_link_libraries(rvclaw_depth_reader PRIVATE ${OPENNI2_LIBRARY})
install(TARGETS rvclaw_depth_reader RUNTIME DESTINATION bin)
